# Test Specification for Music Player System
# Targeted Module: main.py (handle_command)

# User Command Input

Command:
    Empty String.               [error]
    Shortcut (p/s/m).           [single]
    
    /play.                      [property CmdPlay]
    /seek.                      [property CmdSeek]
    /speed.                     [property CmdSpeed]
    
    /tag.add.                   [property CmdTag]
    /import.                    [property CmdImport]
    
    Unknown Command.            [error]

# Arguments Provided

Arguments:
    None.                       [if CmdPlay] [property NoArgs]
    
    One Valid Arg.              [if CmdSpeed] [if CmdSeek] [if CmdImport] [property OneArg]
    
    Two Valid Args.             [if CmdTag] [property TwoArgs]
    
    Missing/Empty.              [if CmdSpeed] [if CmdTag] [if CmdImport] [error]
